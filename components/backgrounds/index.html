CSS 
.bg-blobs {
  background: #0f0f0f;
  position: relative;
  overflow: hidden;
}
.blob {
  position: absolute;
  border-radius: 50%;
  filter: blur(60px);
  opacity: 0.5;
  animation: blobMove 15s ease-in-out infinite;
}
.blob:nth-child(1) {
  width: 200px; height: 200px;
  background: #6366f1;
  top: -50px; left: -50px;
}
.blob:nth-child(2) {
  width: 150px; height: 150px;
  background: #ec4899;
  bottom: -30px; right: -30px;
  animation-delay: -5s;
}
@keyframes blobMove {
  0%, 100% { transform: translate(0, 0) scale(1); }
  33% { transform: translate(30px, 20px) scale(1.1); }
  66% { transform: translate(-20px, -10px) scale(0.9); }
}
