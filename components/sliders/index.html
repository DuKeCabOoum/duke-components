With Tooltip

50 
<div class="slider-tooltip-wrapper">
  <span class="slider-tooltip" id="tooltip">50</span>
  <input type="range" id="slider" value="50">
</div>

/* JavaScript */
slider.addEventListener('input', (e) => {
  tooltip.textContent = e.target.value;
  const percent = e.target.value / 100;
  tooltip.style.left = `calc(${percent * 100}% - ${percent * 20}px)`;
});
