function switchTab(type, index) {
  const container = document.getElementById(`tabs-${type}`);
  const btns = container.querySelectorAll(`.tabs-${type}__btn`);
  const panels = container.querySelectorAll(`.tabs-${type}__panel`);

  btns.forEach(btn => btn.classList.remove(`tabs-${type}__btn--active`));
  panels.forEach(panel => panel.classList.remove(`tabs-${type}__panel--active`));

  btns[index].classList.add(`tabs-${type}__btn--active`);
  panels[index].classList.add(`tabs-${type}__panel--active`);
}
