Dynamic Button

advanced js sensitive 
<button class="btn-dynamic" onclick="handleDynamicAction(this)">
  Dynamic Action
</button>
// ⚠️ ATTENTION : Ce code utilise innerHTML
// Risque XSS si utilisé avec des données utilisateur non sanitizées

function handleDynamicAction(btn) {
  const originalText = btn.textContent;

  // Simule une action asynchrone
  btn.innerHTML = '<span class="loader"></span> Loading...';
  btn.disabled = true;

  setTimeout(() => {
    btn.innerHTML = '✓ Success!';
    btn.classList.add('btn-dynamic--success');

    setTimeout(() => {
      btn.textContent = originalText;
      btn.disabled = false;
      btn.classList.remove('btn-dynamic--success');
    }, 1500);
  }, 2000);
}

// 🔐 VERSION SÉCURISÉE :
// Utiliser textContent au lieu de innerHTML
// ou sanitizer les données avec DOMPurify
